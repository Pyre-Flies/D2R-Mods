cmake_minimum_required(VERSION 3.29)
project(item-roll-ranges VERSION 1.3.1.13 LANGUAGES CXX RC ASM_MASM)
option(ITEM_ROLL_RANGES_COVERAGE_DIAGNOSTICS "Log bounded tooltip coverage evidence for Item Spawner audits" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_DEFAULT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/D2RLoader-PluginSDK")
set(PLUGIN_SDK "${SDK_DEFAULT}" CACHE PATH "D2RLoader PluginSDK")
if(NOT EXISTS "${PLUGIN_SDK}/include/D2RLPlugin/api.h")
    message(FATAL_ERROR "D2RLoader PluginSDK not found at: ${PLUGIN_SDK}")
endif()
add_library(item-roll-ranges SHARED src/plugin.cpp src/version.rc)
target_include_directories(item-roll-ranges PRIVATE "${PLUGIN_SDK}/include")
target_compile_definitions(item-roll-ranges PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(ITEM_ROLL_RANGES_COVERAGE_DIAGNOSTICS)
    target_compile_definitions(item-roll-ranges PRIVATE ITEM_ROLL_RANGES_COVERAGE_DIAGNOSTICS=1)
endif()
target_compile_options(item-roll-ranges PRIVATE /W4 /WX /guard:cf)
target_link_options(item-roll-ranges PRIVATE /guard:cf)
target_link_libraries(item-roll-ranges PRIVATE user32 bcrypt)
set_target_properties(item-roll-ranges PROPERTIES OUTPUT_NAME "Item Roll Ranges" PREFIX "")
enable_testing()
add_executable(range-policy-tests tests/policy_tests.cpp)
target_include_directories(range-policy-tests PRIVATE src)
target_compile_options(range-policy-tests PRIVATE /W4 /WX)
add_test(NAME range-policy COMMAND range-policy-tests)
add_executable(range-adapter-tests tests/adapter_tests.cpp)
target_include_directories(range-adapter-tests PRIVATE "${PLUGIN_SDK}/include")
target_compile_definitions(range-adapter-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(range-adapter-tests PRIVATE /W4 /WX /guard:cf)
target_link_options(range-adapter-tests PRIVATE /guard:cf)
target_link_libraries(range-adapter-tests PRIVATE user32 bcrypt)
add_test(NAME range-adapters COMMAND range-adapter-tests)
add_executable(range-artifact-tests tests/artifact_tests.cpp)
target_include_directories(range-artifact-tests PRIVATE "${PLUGIN_SDK}/include")
target_compile_definitions(range-artifact-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE)
target_compile_options(range-artifact-tests PRIVATE /W4 /WX)
add_dependencies(range-artifact-tests item-roll-ranges)
add_test(NAME range-artifact COMMAND range-artifact-tests "$<TARGET_FILE:item-roll-ranges>")
# Historical JMP-only stack fixture is retained as research, not a loader integration test.

add_executable(range-text-tests tests/range_text_tests.cpp)
target_include_directories(range-text-tests PRIVATE src)
target_compile_options(range-text-tests PRIVATE /W4 /WX)
add_test(NAME range-text COMMAND range-text-tests)
add_executable(range-affix-tests tests/affix_tests.cpp)
target_include_directories(range-affix-tests PRIVATE src)
target_compile_options(range-affix-tests PRIVATE /W4 /WX)
add_test(NAME range-affixes COMMAND range-affix-tests)
