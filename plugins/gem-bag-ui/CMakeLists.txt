cmake_minimum_required(VERSION 3.29)
project(gem-bag-ui VERSION 1.3.1.2 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(D2R_MODS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "D2R-Mods source checkout")
set(SDK "${D2R_MODS_ROOT}/third_party/D2RLoader-PluginSDK")
add_library(gem-bag-ui SHARED src/plugin.cpp src/native.cpp src/catalog.cpp src/geometry.cpp src/rpc.cpp src/server_frame.cpp src/version.rc)
target_include_directories(gem-bag-ui PRIVATE "${SDK}/include" "${D2R_MODS_ROOT}/plugins/controller-qol/src" "${D2R_MODS_ROOT}/plugins/inventory-deposit/src")
target_compile_definitions(gem-bag-ui PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(gem-bag-ui PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/guard:cf;/EHsc>")
target_link_options(gem-bag-ui PRIVATE /guard:cf /PDBALTPATH:%_PDB%)
target_link_libraries(gem-bag-ui PRIVATE bcrypt user32 shell32)
set_target_properties(gem-bag-ui PROPERTIES OUTPUT_NAME "Gem Bag UI" PREFIX "")
enable_testing()
foreach(name IN ITEMS policy artifact deposit)
  add_executable(gem-bag-${name}-tests tests/${name}_tests.cpp)
  target_include_directories(gem-bag-${name}-tests PRIVATE "${SDK}/include" src)
  target_compile_definitions(gem-bag-${name}-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(gem-bag-${name}-tests PRIVATE /W4 /WX /EHsc)
endforeach()
add_dependencies(gem-bag-artifact-tests gem-bag-ui)
add_test(NAME gem-bag-policy COMMAND gem-bag-policy-tests)
add_test(NAME gem-bag-deposit COMMAND gem-bag-deposit-tests)
add_test(NAME gem-bag-artifact COMMAND gem-bag-artifact-tests "$<TARGET_FILE:gem-bag-ui>")
add_executable(gem-bag-native-tests tests/native_tests.cpp src/native.cpp src/catalog.cpp src/server_frame.cpp)
target_include_directories(gem-bag-native-tests PRIVATE "${SDK}/include" src "${D2R_MODS_ROOT}/plugins/controller-qol/src")
target_compile_definitions(gem-bag-native-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX GEMBAG_NATIVE_TESTS)
target_compile_options(gem-bag-native-tests PRIVATE /W4 /WX /EHsc)
target_link_libraries(gem-bag-native-tests PRIVATE bcrypt)
add_test(NAME gem-bag-native COMMAND gem-bag-native-tests)
add_executable(gem-bag-geometry-tests tests/geometry_tests.cpp src/geometry.cpp)
target_include_directories(gem-bag-geometry-tests PRIVATE "${SDK}/include" src "${D2R_MODS_ROOT}/plugins/controller-qol/src" "${D2R_MODS_ROOT}/plugins/inventory-deposit/src")
target_compile_definitions(gem-bag-geometry-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(gem-bag-geometry-tests PRIVATE /W4 /WX /EHsc)
target_link_libraries(gem-bag-geometry-tests PRIVATE user32)
add_test(NAME gem-bag-geometry COMMAND gem-bag-geometry-tests)
add_executable(gem-bag-ui-lifecycle-tests tests/ui_lifecycle_tests.cpp src/catalog.cpp src/rpc.cpp src/server_frame.cpp)
target_include_directories(gem-bag-ui-lifecycle-tests PRIVATE "${SDK}/include" src "${D2R_MODS_ROOT}/plugins/controller-qol/src" "${D2R_MODS_ROOT}/plugins/inventory-deposit/src")
target_compile_definitions(gem-bag-ui-lifecycle-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(gem-bag-ui-lifecycle-tests PRIVATE /W4 /WX /EHsc)
target_link_libraries(gem-bag-ui-lifecycle-tests PRIVATE user32 shell32 bcrypt)
add_test(NAME gem-bag-ui-lifecycle COMMAND gem-bag-ui-lifecycle-tests "${CMAKE_CURRENT_BINARY_DIR}/startup-layout.json")
add_test(NAME gem-bag-host-load COMMAND gem-bag-ui-lifecycle-tests --host)
add_test(NAME gem-bag-generic-picker COMMAND gem-bag-ui-lifecycle-tests --portable)
add_executable(gem-bag-rpc-tests tests/rpc_tests.cpp src/rpc.cpp)
target_include_directories(gem-bag-rpc-tests PRIVATE "${SDK}/include" src)
target_compile_definitions(gem-bag-rpc-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(gem-bag-rpc-tests PRIVATE /W4 /WX /EHsc)
target_link_libraries(gem-bag-rpc-tests PRIVATE bcrypt)
add_test(NAME gem-bag-rpc COMMAND gem-bag-rpc-tests)
add_executable(gem-bag-frame-tests tests/frame_bridge_tests.cpp)
target_include_directories(gem-bag-frame-tests PRIVATE src "${D2R_MODS_ROOT}/plugins/controller-qol/src")
target_compile_definitions(gem-bag-frame-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(gem-bag-frame-tests PRIVATE /W4 /WX /EHsc)
add_test(NAME gem-bag-frame COMMAND gem-bag-frame-tests)
add_executable(gem-bag-admission-tests tests/admission_tests.cpp src/native.cpp src/catalog.cpp src/server_frame.cpp)
target_include_directories(gem-bag-admission-tests PRIVATE "${SDK}/include" src "${D2R_MODS_ROOT}/plugins/controller-qol/src")
target_compile_definitions(gem-bag-admission-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(gem-bag-admission-tests PRIVATE /W4 /WX /EHsc)
target_link_libraries(gem-bag-admission-tests PRIVATE bcrypt)

target_include_directories(gem-bag-catalog-tests PRIVATE "${SDK}/include" src)
target_compile_definitions(gem-bag-catalog-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
