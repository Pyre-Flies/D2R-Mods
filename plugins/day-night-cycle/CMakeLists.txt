cmake_minimum_required(VERSION 3.29)
project(day-night-cycle VERSION 1.3.1.1 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(D2R_MODS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Reviewed public source checkout")
set(SDK "${D2R_MODS_ROOT}/third_party/D2RLoader-PluginSDK" CACHE PATH "D2RLoader PluginSDK")
find_package(Python3 REQUIRED COMPONENTS Interpreter)

add_library(day-night-cycle SHARED src/plugin.cpp src/version.rc vendor/src/native_commands.cpp vendor/src/image_scan.cpp)
include("${SDK}/cmake/D2RLPluginEmbedConfig.cmake")
d2rlplugin_embed_config(TARGET day-night-cycle FILE "${CMAKE_CURRENT_SOURCE_DIR}/day-night-cycle.toml")
target_include_directories(day-night-cycle PRIVATE "${SDK}/include" src vendor/include)
target_compile_definitions(day-night-cycle PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(day-night-cycle PRIVATE /W4 /WX /guard:cf)
target_link_options(day-night-cycle PRIVATE /guard:cf /PDBALTPATH:%_PDB%)
target_link_libraries(day-night-cycle PRIVATE bcrypt)
set_target_properties(day-night-cycle PROPERTIES OUTPUT_NAME "Day Night Cycle" PREFIX "")

enable_testing()
foreach(test IN ITEMS artifact native_path guards command)
  add_executable(day-night-${test}-tests tests/${test}_tests.cpp)
  target_include_directories(day-night-${test}-tests PRIVATE "${SDK}/include" src vendor/include)
  target_compile_definitions(day-night-${test}-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE)
  target_compile_options(day-night-${test}-tests PRIVATE /W4 /WX /guard:cf)
  target_link_options(day-night-${test}-tests PRIVATE /guard:cf)
endforeach()
target_sources(day-night-command-tests PRIVATE vendor/src/image_scan.cpp)
target_link_libraries(day-night-command-tests PRIVATE bcrypt)
add_dependencies(day-night-artifact-tests day-night-cycle)
target_link_libraries(day-night-artifact-tests PRIVATE bcrypt)
add_test(NAME day-night-artifact COMMAND day-night-artifact-tests "$<TARGET_FILE:day-night-cycle>")
add_test(NAME day-night-native-path COMMAND day-night-native_path-tests)
add_test(NAME day-night-guards COMMAND day-night-guards-tests)
add_test(NAME day-night-command COMMAND day-night-command-tests)
add_test(NAME day-night-profile COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_profile.py" --check)
