cmake_minimum_required(VERSION 3.29)
project(map-assistance VERSION 1.3.1 LANGUAGES CXX RC)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_DEFAULT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/D2RLoader-PluginSDK")
if(NOT EXISTS "${SDK_DEFAULT}/include/D2RLPlugin/api.h" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/sdk/include/D2RLPlugin/api.h")
    set(SDK_DEFAULT "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
endif()
set(PLUGIN_SDK "${SDK_DEFAULT}" CACHE PATH "D2RLoader PluginSDK")
if(NOT EXISTS "${PLUGIN_SDK}/include/D2RLPlugin/api.h")
    message(FATAL_ERROR "D2RLoader PluginSDK not found at: ${PLUGIN_SDK}")
endif()

add_library(map-assistance SHARED src/plugin.cpp src/version.rc)
include("${PLUGIN_SDK}/cmake/D2RLPluginEmbedConfig.cmake")
d2rlplugin_embed_config(TARGET map-assistance FILE "${CMAKE_CURRENT_SOURCE_DIR}/map-assistance.toml")
target_include_directories(map-assistance PRIVATE "${PLUGIN_SDK}/include")
target_compile_definitions(map-assistance PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(map-assistance PRIVATE /W4 /WX /guard:cf)
target_link_options(map-assistance PRIVATE /guard:cf)
set_target_properties(map-assistance PROPERTIES OUTPUT_NAME "Map Assistance" PREFIX "")

enable_testing()
add_executable(map-assistance-config-tests tests/map_config_tests.cpp)
target_include_directories(map-assistance-config-tests PRIVATE src)
target_compile_options(map-assistance-config-tests PRIVATE /W4 /WX)
add_test(NAME map-assistance-config COMMAND map-assistance-config-tests "${CMAKE_CURRENT_SOURCE_DIR}/map-assistance.toml")

add_executable(map-assistance-artifact-tests tests/artifact_tests.cpp)
target_include_directories(map-assistance-artifact-tests PRIVATE "${PLUGIN_SDK}/include")
target_compile_definitions(map-assistance-artifact-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE)
target_compile_options(map-assistance-artifact-tests PRIVATE /W4 /WX)
add_dependencies(map-assistance-artifact-tests map-assistance)
add_test(NAME map-assistance-artifact COMMAND map-assistance-artifact-tests "$<TARGET_FILE:map-assistance>")
