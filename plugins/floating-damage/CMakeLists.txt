cmake_minimum_required(VERSION 3.29)
project(floating-damage VERSION 1.3.1.12 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PLUGIN_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/D2RLoader-PluginSDK" CACHE PATH "D2RLoader PluginSDK")
add_library(floating-damage SHARED src/plugin.cpp src/version.rc vendor/src/native_commands.cpp vendor/src/image_scan.cpp)
include("${PLUGIN_SDK}/cmake/D2RLPluginEmbedConfig.cmake")
d2rlplugin_embed_config(TARGET floating-damage FILE "${CMAKE_CURRENT_SOURCE_DIR}/floating-damage.toml")
target_include_directories(floating-damage PRIVATE "${PLUGIN_SDK}/include" vendor/include)
target_compile_definitions(floating-damage PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(floating-damage PRIVATE /W4 /WX /guard:cf)
target_link_options(floating-damage PRIVATE /guard:cf)
target_link_libraries(floating-damage PRIVATE bcrypt)
set_target_properties(floating-damage PROPERTIES OUTPUT_NAME "Floating Damage" PREFIX "")
enable_testing()
add_executable(floating-damage-model-tests tests/model_tests.cpp)
target_include_directories(floating-damage-model-tests PRIVATE src)
target_compile_options(floating-damage-model-tests PRIVATE /W4 /WX)
add_test(NAME floating-damage-model COMMAND floating-damage-model-tests)
add_executable(floating-damage-reader-tests tests/reader_tests.cpp)
target_include_directories(floating-damage-reader-tests PRIVATE src)
target_compile_definitions(floating-damage-reader-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(floating-damage-reader-tests PRIVATE /W4 /WX)
target_link_libraries(floating-damage-reader-tests PRIVATE bcrypt)
add_test(NAME floating-damage-reader COMMAND floating-damage-reader-tests)
add_executable(floating-damage-artifact-tests tests/artifact_tests.cpp)
target_include_directories(floating-damage-artifact-tests PRIVATE "${PLUGIN_SDK}/include" src)
target_compile_definitions(floating-damage-artifact-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE)
target_compile_options(floating-damage-artifact-tests PRIVATE /W4 /WX)
add_dependencies(floating-damage-artifact-tests floating-damage)
add_test(NAME floating-damage-artifact COMMAND floating-damage-artifact-tests "$<TARGET_FILE:floating-damage>")
add_executable(floating-damage-element-tests tests/element_tests.cpp)
target_include_directories(floating-damage-element-tests PRIVATE src)
target_compile_options(floating-damage-element-tests PRIVATE /W4 /WX)
add_test(NAME floating-damage-elements COMMAND floating-damage-element-tests)
add_executable(floating-damage-critical-tests tests/critical_tests.cpp)
target_include_directories(floating-damage-critical-tests PRIVATE src)
target_compile_options(floating-damage-critical-tests PRIVATE /W4 /WX)
add_test(NAME floating-damage-critical COMMAND floating-damage-critical-tests)
add_executable(floating-damage-combat-tests tests/combat_tests.cpp vendor/src/native_commands.cpp vendor/src/image_scan.cpp)
target_include_directories(floating-damage-combat-tests PRIVATE "${PLUGIN_SDK}/include" src vendor/include)
target_compile_definitions(floating-damage-combat-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(floating-damage-combat-tests PRIVATE /W4 /WX)
target_link_libraries(floating-damage-combat-tests PRIVATE bcrypt)
add_test(NAME floating-damage-combat COMMAND floating-damage-combat-tests)
foreach(fixture IN ITEMS native-chat-fixture-a native-chat-fixture-b)
    add_library(${fixture} SHARED tests/native_commands_module.cpp vendor/src/image_scan.cpp)
    target_include_directories(${fixture} PRIVATE "${PLUGIN_SDK}/include" vendor/include)
    target_compile_options(${fixture} PRIVATE /W4 /WX)
endforeach()
add_executable(floating-damage-chat-tests tests/native_commands_tests.cpp)
target_include_directories(floating-damage-chat-tests PRIVATE "${PLUGIN_SDK}/include" vendor/include)
target_compile_options(floating-damage-chat-tests PRIVATE /W4 /WX)
add_dependencies(floating-damage-chat-tests native-chat-fixture-a native-chat-fixture-b)
add_test(NAME floating-damage-chat COMMAND floating-damage-chat-tests "$<TARGET_FILE:native-chat-fixture-a>" "$<TARGET_FILE:native-chat-fixture-b>")
