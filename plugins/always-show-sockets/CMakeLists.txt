cmake_minimum_required(VERSION 3.29)
project(always-show-sockets LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PLUGIN_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/D2RLoader-PluginSDK")
add_library(always-show-sockets SHARED src/plugin.cpp src/version.rc vendor/src/native_commands.cpp vendor/src/image_scan.cpp)
target_include_directories(always-show-sockets PRIVATE "${PLUGIN_SDK}/include" vendor/include)
target_compile_definitions(always-show-sockets PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(always-show-sockets PRIVATE /W4 /WX /guard:cf)
target_link_options(always-show-sockets PRIVATE /guard:cf)
target_link_libraries(always-show-sockets PRIVATE bcrypt)
set_target_properties(always-show-sockets PROPERTIES OUTPUT_NAME "Always Show Sockets" PREFIX "")
enable_testing()
add_executable(socket-artifact-tests tests/artifact_tests.cpp)
target_include_directories(socket-artifact-tests PRIVATE "${PLUGIN_SDK}/include")
target_compile_definitions(socket-artifact-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE)
target_compile_options(socket-artifact-tests PRIVATE /W4 /WX)
add_dependencies(socket-artifact-tests always-show-sockets)
add_test(NAME socket-artifact COMMAND socket-artifact-tests "$<TARGET_FILE:always-show-sockets>")
add_executable(socket-command-tests tests/command_tests.cpp vendor/src/native_commands.cpp vendor/src/image_scan.cpp)
target_include_directories(socket-command-tests PRIVATE "${PLUGIN_SDK}/include" vendor/include)
target_compile_definitions(socket-command-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(socket-command-tests PRIVATE /W4 /WX)
target_link_libraries(socket-command-tests PRIVATE bcrypt)
add_test(NAME socket-command COMMAND socket-command-tests)
