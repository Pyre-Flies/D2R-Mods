cmake_minimum_required(VERSION 3.29)
project(always-show-sockets VERSION 1.3.1.1 LANGUAGES CXX RC ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PLUGIN_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/D2RLoader-PluginSDK")
add_library(always-show-sockets SHARED src/plugin.cpp src/version.rc src/post_icon_gate.asm src/scale_gate.asm src/circle_gate.asm vendor/src/native_commands.cpp vendor/src/image_scan.cpp)
include("${PLUGIN_SDK}/cmake/D2RLPluginEmbedConfig.cmake")
d2rlplugin_embed_config(TARGET always-show-sockets FILE "${CMAKE_CURRENT_SOURCE_DIR}/always-show-sockets.toml")
target_include_directories(always-show-sockets PRIVATE "${PLUGIN_SDK}/include" vendor/include)
target_compile_definitions(always-show-sockets PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(always-show-sockets PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/guard:cf>")
target_link_options(always-show-sockets PRIVATE /guard:cf)
target_link_libraries(always-show-sockets PRIVATE bcrypt)
set_target_properties(always-show-sockets PROPERTIES OUTPUT_NAME "Always Show Sockets" PREFIX "")
enable_testing()
add_executable(socket-artifact-tests tests/artifact_tests.cpp)
target_include_directories(socket-artifact-tests PRIVATE "${PLUGIN_SDK}/include")
target_compile_definitions(socket-artifact-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE)
target_compile_options(socket-artifact-tests PRIVATE /W4 /WX)
add_dependencies(socket-artifact-tests always-show-sockets)
add_test(NAME socket-artifact COMMAND socket-artifact-tests "$<TARGET_FILE:always-show-sockets>")
add_executable(socket-command-tests tests/command_tests.cpp src/post_icon_gate.asm src/scale_gate.asm src/circle_gate.asm tests/post_icon_fixture.asm tests/scale_fixture.asm tests/circle_fixture.asm vendor/src/image_scan.cpp)
target_include_directories(socket-command-tests PRIVATE "${PLUGIN_SDK}/include" vendor/include)
target_compile_definitions(socket-command-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(socket-command-tests PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX>")
target_link_libraries(socket-command-tests PRIVATE bcrypt)
add_test(NAME socket-command COMMAND socket-command-tests)
add_executable(socket-scale-tests tests/scale_tests.cpp)
target_compile_options(socket-scale-tests PRIVATE /W4 /WX)
add_test(NAME socket-scale-policy COMMAND socket-scale-tests)
add_executable(socket-ring-tests tests/ring_tests.cpp)
target_include_directories(socket-ring-tests PRIVATE "${PLUGIN_SDK}/include")
target_compile_definitions(socket-ring-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(socket-ring-tests PRIVATE /W4 /WX)
add_test(NAME socket-ring COMMAND socket-ring-tests)
